class BrowserManager:
    def __init__(self, headless=True):
        options = webdriver.ChromeOptions()
        if headless:
            options.add_argument('--headless')
        options.add_argument('--no-sandbox')
        options.add_argument('--disable-dev-shm-usage')

        self.wait = WebDriverWait(self.driver, 10)
    
    def get(self, url):
        self.driver.get(url)
    
    def find_element(self, by, value):
        return self.wait.until(EC.presence_of_element_located((by, value)))
    
    def quit(self):
        self.driver.quit()

# src/automation/arxiv_submitter.py
from browser_manager import BrowserManager
from selenium.webdriver.common.by import By
import time

class ArxivSubmitter:
    def __init__(self):
        self.base_url = "https://arxiv.org/user/login"
        self.browser = BrowserManager(headless=False)
    
    def login(self, username, password):
        self.browser.get(self.base_url)
        
        # 填写登录表单
        username_field = self.browser.find_element(By.ID, "username")
        password_field = self.browser.find_element(By.ID, "password")
        
        username_field.send_keys(username)
        password_field.send_keys(password)
        
        # 点击登录
        login_btn = self.browser.find_element(By.CSS_SELECTOR, "button[type='submit']")
        login_btn.click()
        time.sleep(2)
    
    def submit_paper(self, paper_data):
        # 导航到投稿页面
        self.browser.get("https://arxiv.org/submit")
        
        # 填写标题
        title_field = self.browser.find_element(By.ID, "title")
        title_field.send_keys(paper_data['title'])
        
        # 填写作者
        authors_field = self.browser.find_element(By.ID, "authors")
        authors_field.send_keys(", ".join(paper_data['authors']))
        
        # 填写摘要
        abstract_field = self.browser.find_element(By.ID, "abstract")
        abstract_field.send_keys(paper_data['abstract'])
        
        # 上传PDF
        file_input = self.browser.find_element(By.ID, "file")
        file_input.send_keys(paper_data['file_path'])
        
        # 提交
        submit_btn = self.browser.find_element(By.ID, "submit-button")
        submit_btn.click()
        
        # 获取稿件编号
        time.sleep(3)
        submission_id = self.browser.find_element(By.CLASS_NAME, "submission-id").text
        
        return submission_id
    
    def check_status(self, submission_id):
        self.browser.get(f"https://arxiv.org/user/submissions/{submission_id}")
        status_element = self.browser.find_element(By.CLASS_NAME, "status")
        return status_element.text
    
    def close(self):
        self.browser.quit()
```

# src/backend/security.py
from cryptography.fernet import Fernet
import os
from dotenv import load_dotenv

load_dotenv()

class CredentialManager:
    def __init__(self):
        # 从环境变量读取密钥，如果不存在则生成
        key = os.getenv('ENCRYPTION_KEY')
        if not key:
            key = Fernet.generate_key()
            # 保存到.env文件
            with open('.env', 'a') as f:
                f.write(f"\nENCRYPTION_KEY={key.decode()}")
        else:
            key = key.encode()
        
        self.cipher = Fernet(key)
    
    def encrypt_password(self, password: str) -> str:
        return self.cipher.encrypt(password.encode()).decode()
    
    def decrypt_password(self, encrypted_password: str) -> str:
        return self.cipher.decrypt(encrypted_password.encode()).decode()
```

    db.refresh(paper)
    db.close()

@app.post("/api/submissions/create")
async def create_submission(
    paper_id: int,
    journal_name: str,
    username: str,
    password: str
):
    # 加密并保存凭证
    encrypted_pwd = credential_manager.encrypt_password(password)
    
    db = SessionLocal()
    credential = Credential(
        journal_name=journal_name,
        username=username,
        encrypted_password=encrypted_pwd
    )
    db.add(credential)
    
    # 创建投稿记录
    submission = Submission(
        paper_id=paper_id,
        journal_name=journal_name,
        status="submitting"
    )
    db.add(submission)
    db.commit()
    
    # 执行自动投稿（异步）
    try:
        if journal_name.lower() == "arxiv":
            submitter = ArxivSubmitter()
            paper = db.query(Paper).filter(Paper.id == paper_id).first()
            
            submitter.login(username, password)
            submission_id = submitter.submit_paper({
                'title': paper.title,
                'authors': json.loads(paper.authors),
                'abstract': paper.abstract,
                'file_path': paper.file_path
            })
            
            submission.submission_id = submission_id
            submission.status = "submitted"
            submitter.close()
    except Exception as e:
        submission.status = "failed"
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        db.commit()
        db.close()
    
    return {"submission_id": submission.id, "status": submission.status}

@app.get("/api/submissions/{submission_id}/status")
async def get_submission_status(submission_id: int):

    return {
        "submission_id": submission.id,
        "journal": submission.journal_name,

from sklearn.feature_extraction.text import TfidfVectorizer
import PyPDF2

class KeywordExtractor:
    def __init__(self):
        self.vectorizer = TfidfVectorizer(max_features=10, stop_words='english')
    
    def extract_from_pdf(self, pdf_path):
        # 读取PDF文本
        with open(pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            text = ""
            for page in reader.pages[:5]:  # 只读前5页
                text += page.extract_text()
        
        return self.extract_from_text(text)
    
    def extract_from_text(self, text):
        # 使用TF-IDF提取关键词
        try:
            tfidf_matrix = self.vectorizer.fit_transform([text])
            feature_names = self.vectorizer.get_feature_names_out()
            return list(feature_names)
        except:
            return []

**3. 期刊匹配算法 (1.5h)**
```python
# src/ml_models/journal_recommender.py
import json
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.feature_extraction.text import TfidfVectorizer

class JournalRecommender:
    def __init__(self, journal_db_path="../../config/journals/database.json"):
        with open(journal_db_path, 'r') as f:
            self.journals = json.load(f)
        
        self.vectorizer = TfidfVectorizer()
        
        # 预处理期刊关键词
        journal_texts = [" ".join(j['keywords']) for j in self.journals]
        self.journal_vectors = self.vectorizer.fit_transform(journal_texts)
    
    def recommend(self, paper_keywords, top_k=5):
        # 将论文关键词转换为向量
        paper_text = " ".join(paper_keywords)
        paper_vector = self.vectorizer.transform([paper_text])
        
        # 计算相似度
        similarities = cosine_similarity(paper_vector, self.journal_vectors)[0]
        
        # 综合评分（相似度40% + 影响因子30% + 审稿速度20% + 接收率10%）
        scores = []
        for i, journal in enumerate(self.journals):
            match_score = similarities[i] * 0.4
            if_score = min(journal['impact_factor'] / 50, 1) * 0.3
            speed_score = (1 - min(journal['review_time_days'] / 180, 1)) * 0.2
            acceptance_score = journal['acceptance_rate'] * 0.1
            
            total_score = match_score + if_score + speed_score + acceptance_score
            scores.append({
                'journal': journal['name'],
                'score': total_score,
                'match_score': similarities[i],
                'impact_factor': journal['impact_factor'],
                'review_time': journal['review_time_days'],
                'acceptance_rate': journal['acceptance_rate']
            })
        
        # 排序并返回Top K
        scores.sort(key=lambda x: x['score'], reverse=True)
        return scores[:top_k]
```

import React, { useEffect, useState } from 'react';
import { Table, Tag, Card, Statistic, Row, Col } from 'antd';
import { CheckCircleOutlined, SyncOutlined, ClockCircleOutlined } from '@ant-design/icons';
import axios from 'axios';

function Dashboard() {
  const [submissions, setSubmissions] = useState([]);
  const [stats, setStats] = useState({ total: 0, submitted: 0, reviewing: 0 });

  useEffect(() => {
    fetchSubmissions();
  }, []);

  const fetchSubmissions = async () => {
    try {
      const response = await axios.get('http://localhost:8000/api/submissions');
      setSubmissions(response.data);
      
      // 计算统计数据
      const total = response.data.length;
      const submitted = response.data.filter(s => s.status === 'submitted').length;
      const reviewing = response.data.filter(s => s.status === 'under_review').length;
      setStats({ total, submitted, reviewing });
    } catch (error) {
      console.error('获取投稿列表失败:', error);
    }
  };

  const columns = [
    {
      title: '论文标题',
      dataIndex: 'paper_title',
      key: 'title',
    },
    {
      title: '期刊',
      dataIndex: 'journal_name',
      key: 'journal',
    },
    {
      title: '状态',
      dataIndex: 'status',
      key: 'status',
      render: (status) => {
        const statusMap = {
          'submitted': { color: 'blue', text: '已提交', icon: <CheckCircleOutlined /> },
          'under_review': { color: 'orange', text: '审稿中', icon: <SyncOutlined spin /> },
          'pending': { color: 'default', text: '待处理', icon: <ClockCircleOutlined /> },
        };
        const config = statusMap[status] || statusMap['pending'];
        return <Tag color={config.color} icon={config.icon}>{config.text}</Tag>;
      },
    },
    {
      title: '提交时间',
      dataIndex: 'created_at',
      key: 'created_at',
      render: (date) => new Date(date).toLocaleDateString(),
    },
  ];

  return (
    <div style={{ padding: 20 }}>
      <h1>投稿仪表盘</h1>
      
      <Row gutter={16} style={{ marginBottom: 20 }}>
        <Col span={8}>
          <Card>
            <Statistic title="总投稿数" value={stats.total} />
          </Card>
        </Col>
        <Col span={8}>
          <Card>
            <Statistic title="已提交" value={stats.submitted} valueStyle={{ color: '#3f8600' }} />
          </Card>
        </Col>
        <Col span={8}>
          <Card>
            <Statistic title="审稿中" value={stats.reviewing} valueStyle={{ color: '#cf1322' }} />
          </Card>
        </Col>
      </Row>

      <Card>
        <Table columns={columns} dataSource={submissions} rowKey="id" />
      </Card>
    </div>
  );
}

export default Dashboard;
```

# tests/test_e2e.py
import requests
import time

BASE_URL = "http://localhost:8000"

def test_full_workflow():
    print("🧪 开始端到端测试...")

    print("1️⃣ 上传论文...")

            "title": "Deep Learning for Paper Submission",
            "authors": '["Alice", "Bob"]',
            "abstract": "This paper presents an automated system..."
        }
        response = requests.post(f"{BASE_URL}/api/papers/upload", files=files, data=data)
        assert response.status_code == 200
        paper_id = response.json()['paper_id']
        print(f"✅ 论文上传成功，ID: {paper_id}")
    
    # 2. 获取期刊推荐
    print("2️⃣ 获取期刊推荐...")
    response = requests.post(f"{BASE_URL}/api/journals/recommend", json={"paper_id": paper_id})
    assert response.status_code == 200
    recommendations = response.json()['recommendations']
    print(f"✅ 推荐期刊: {[r['journal'] for r in recommendations]}")
    
    # 3. 创建投稿
    print("3️⃣ 创建投稿...")
    data = {
        "paper_id": paper_id,
        "journal_name": "arxiv",
        "username": "test_user",
        "password": "test_password"
    }
    response = requests.post(f"{BASE_URL}/api/submissions/create", json=data)
    assert response.status_code == 200
    submission_id = response.json()['submission_id']
    print(f"✅ 投稿创建成功，ID: {submission_id}")
    
    # 4. 检查投稿状态
    print("4️⃣ 检查投稿状态...")

    response = requests.get(f"{BASE_URL}/api/submissions/{submission_id}/status")

    print(f"✅ 当前状态: {status}")
    
    print("\n🎉 所有测试通过！")

if __name__ == "__main__":
    test_full_workflow()

# PaperSubmit AI - 使用指南

## 快速开始

### 1. 启动后端服务
\`\`\`bash
cd src/backend
source venv/bin/activate
python main.py
\`\`\`

### 2. 启动前端服务
\`\`\`bash
cd src/frontend/papersubmit-ui
npm start
\`\`\`

### 3. 访问应用
打开浏览器访问: http://localhost:3000

## 功能说明

### 上传论文
1. 点击"上传论文"菜单
2. 拖拽或选择PDF文件
3. 填写论文标题、作者、摘要
4. 点击"上传并获取期刊推荐"

### 查看推荐期刊
系统会自动分析论文内容，推荐最匹配的5个期刊，包括：
- 匹配度评分
- 影响因子
- 平均审稿时间
- 接收率

### 自动投稿
1. 选择目标期刊
2. 输入投稿系统账号密码（加密存储）
3. 确认投稿
4. 系统自动完成投稿流程

### 跟踪进度
在仪表盘查看所有投稿状态，系统每天自动检查并更新。

## 注意事项
- 目前仅支持arXiv自动投稿
- 其他期刊系统正在开发中
- 请确保账号密码正确，避免被封号
```
